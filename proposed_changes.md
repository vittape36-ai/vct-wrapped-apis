## 1. Database Lookup for Multi-Tenant API Keys

Currently, `src/middleware/auth.js` checks against a single hardcoded key. Since the project already uses `ioredis` and has a `cache.js` utility, the absolute simplest solution is to store and lookup API keys in Redis.

### Solution

**1. Store Keys in Redis (Setup):**
Keys can be stored as JSON strings against their hash:
```javascript
// Example data structure in Redis:
// SET auth:apikey:vct_live_123 '{"plan": "premium", "rateLimit": 100}'
```

**2. Update `src/middleware/auth.js`:**
Replace the hardcoded check with an async Redis lookup.

```javascript
import { fail } from '../utils/response.js';
import { cacheGet } from '../utils/cache.js';

export async function authenticate(req, res, next) {
  const key = req.headers['x-api-key'];
  if (!key) return fail(res, 401, 'Missing x-api-key header');

  try {
    // Attempt to fetch the key details from Redis
    const tenantData = await cacheGet(`auth:apikey:${key}`);
    
    if (!tenantData) {
      return fail(res, 403, 'Invalid or revoked API key');
    }

    // Attach caller identity
    req.vctCaller = { 
      keyId: `${key.slice(0, 8)}...`, 
      plan: tenantData.plan 
    };
    
    next();
  } catch (error) {
    return fail(res, 500, 'Authentication service unavailable');
  }
}
```
*Note: Ensure the Express route handler using this middleware properly catches async errors, or use an async wrapper.*

## 2. Webhook Receiver for Pay+

Razorpay requires a POST endpoint that verifies a cryptographic signature to ensure webhooks are legitimate.

### Solution

**1. Update `src/routes/pay.routes.js`:**
Add a new route for webhooks.

```javascript
import express from 'express';
import crypto from 'crypto';
import { config } from '../config/env.js';

const router = express.Router();

router.post('/webhook', express.raw({ type: 'application/json' }), (req, res) => {
  const secret = config.pay.webhookSecret; // Add this to env.js
  const signature = req.headers['x-razorpay-signature'];
  
  // Verify signature
  const expectedSignature = crypto
    .createHmac('sha256', secret)
    .update(req.body)
    .digest('hex');

  if (expectedSignature !== signature) {
    return res.status(400).send('Invalid signature');
  }

  // Parse body after verification
  const event = JSON.parse(req.body);
  
  if (event.event === 'payout.processed') {
    // TODO: Update internal ledger/db
    console.log('Payout succeeded for:', event.payload.payout.entity.id);
  }

  res.status(200).send('OK');
});

export default router;
```

## 3. WebSocket Streaming for LLM

To stream tokens back to the client as they are generated by OpenAI/Gemini, rather than waiting for the entire block.

### Solution

Instead of pulling in heavy dependencies like `socket.io`, use the native Server-Sent Events (SSE) standard which works flawlessly over standard HTTP.

**1. Update `src/services/llm.service.js` to return a stream:**
```javascript
export async function streamOpenAI(messages, model) {
  const client = new OpenAI({ apiKey: openaiVault.next() });
  // Pass stream: true
  return client.chat.completions.create({
    model: model,
    messages,
    stream: true,
  });
}
```

**2. Update `src/routes/llm.routes.js` to handle SSE:**
```javascript
router.post('/chat/stream', async (req, res) => {
  res.setHeader('Content-Type', 'text/event-stream');
  res.setHeader('Cache-Control', 'no-cache');
  res.setHeader('Connection', 'keep-alive');

  const stream = await streamOpenAI(req.body.messages, req.body.model);

  for await (const chunk of stream) {
    const content = chunk.choices[0]?.delta?.content || '';
    if (content) {
      res.write(`data: ${JSON.stringify({ text: content })}\n\n`);
    }
  }
  res.write('data: [DONE]\n\n');
  res.end();
});
```

## 4. gRPC Bindings

To allow high-performance internal microservices to communicate with VCT APIs using protocol buffers.

### Solution

**1. Create a Proto File (`src/proto/llm.proto`):**
```protobuf
syntax = "proto3";

package vct.llm;

service LlmService {
  rpc Chat (ChatRequest) returns (ChatResponse);
}

message ChatRequest {
  string model = 1;
  string message = 2;
}

message ChatResponse {
  string content = 1;
}
```

**2. Add `grpc` Server logic (in `src/server.js`):**
```bash
npm install @grpc/grpc-js @grpc/proto-loader
```
```javascript
import * as grpc from '@grpc/grpc-js';
import * as protoLoader from '@grpc/proto-loader';
import { chat } from './services/llm.service.js';

const packageDefinition = protoLoader.loadSync('src/proto/llm.proto');
const proto = grpc.loadPackageDefinition(packageDefinition).vct.llm;

const server = new grpc.Server();

server.addService(proto.LlmService.service, {
  Chat: async (call, callback) => {
    try {
      const result = await chat({ 
        messages: [{ role: 'user', content: call.request.message }] 
      });
      callback(null, { content: result.content });
    } catch (err) {
      callback({ code: grpc.status.INTERNAL, details: err.message });
    }
  }
});

server.bindAsync('0.0.0.0:50051', grpc.ServerCredentials.createInsecure(), () => {
  console.log('gRPC server running on port 50051');
});
```

## 6. ClickHouse Analytics Pipeline

Routing logs out of standard output and into ClickHouse for aggregate reporting.

### Solution

Since the project already uses `winston` for logging, we simply write a custom Winston transport that buffers log entries and fires a bulk HTTP POST request to ClickHouse every few seconds.

```javascript
import Transport from 'winston-transport';

export class ClickHouseTransport extends Transport {
  constructor(opts) {
    super(opts);
    this.buffer = [];
    setInterval(() => this.flush(), 5000); // Flush every 5s
  }

  log(info, callback) {
    this.buffer.push(info);
    callback();
  }

  async flush() {
    if (this.buffer.length === 0) return;
    const data = this.buffer.splice(0, this.buffer.length);
    // Send standard HTTP POST to ClickHouse JSONEachRow endpoint
    await fetch('http://clickhouse-server:8123/?query=INSERT INTO logs FORMAT JSONEachRow', {
      method: 'POST',
      body: data.map(d => JSON.stringify(d)).join('\n')
    });
  }
}
```

## 7. Admin Dashboard UI

A lightweight dashboard to manage the Redis-backed API keys and view basic usage stats.

### Solution 

Do not set up a massive separate React/Next.js infrastructure. Use the existing Express server to serve a static, single-page frontend.

**1. Build Frontend:**
Create a simple `index.html` in a `public/` directory using vanilla JS and TailwindCSS via CDN (no build step needed). 

**2. Serve via Express:**
```javascript
// In src/server.js
import express from 'express';

app.use('/admin', express.static('public')); 
```

The frontend can interact with a new protected `/admin/v1/keys` REST endpoint to GET/POST to the Redis store, completing the full management loop in < 100 lines of code.
